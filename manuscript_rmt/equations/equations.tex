% LaTeX source of the manuscript equations (for journal typesetting)

% eq1_mp_edges
\begin{equation}
\lambda_{\pm} = \sigma^{2}\left(1 \pm \sqrt{\gamma}\right)^{2}
\end{equation}

% eq2_tw_statistic
\begin{equation}
s_i = \frac{N\lambda_i/\sigma^{2} - \mu}{\sigma_{TW}},\quad \mu = \left(\sqrt{N-1} + \sqrt{p-i}\right)^{2},\quad \sigma_{TW} = \left(\sqrt{N-1} + \sqrt{p-i}\right)\left(\frac{1}{\sqrt{N-1}} + \frac{1}{\sqrt{p-i}}\right)^{1/3}
\end{equation}

% eq3_noise_fixed_point
\begin{equation}
\sigma^{2} = \frac{p}{p - k + \sum_{i \leq k} \ell_i / \sigma^{2}}
\end{equation}

% eq4_reliability
\begin{equation}
\psi^{2} = \frac{1 - \gamma/(\ell - 1)^{2}}{1 + \gamma/(\ell - 1)}
\end{equation}

% eq5_agreement_D
\begin{equation}
D = \frac{2\,\Vert U_{\mathrm{ref}}^{\top} V \Vert_F^{2}}{K_{\mathrm{ref}} + k},\qquad D = 0\ \mathrm{if}\ k = 0
\end{equation}
